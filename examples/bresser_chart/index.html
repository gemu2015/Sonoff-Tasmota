    Bresser Weather Station + 24h Google Charts (CC1101 868 MHz)

    // @name: Bresser Weather Station
// Bresser Weather Station + 24h Google Charts (CC1101 868 MHz)
// Full Bresser 5/6/7-in-1 receiver with 6 ring-buffer charts
// 288 points per chart (every 5 min, 24 hours)
// Uses WebChart() for automatic Google Charts rendering
// Uses persist keyword for automatic chart data save/restore across reboots
// Globals: 6 × 288 = 1728 slots (of dynamic max on ESP32)

// Uncomment to simulate sensor data (no CC1101 hardware needed)
//#define SIMULATE 1
#ifdef SIMULATE
#define STORE_INTERVAL 10   // fast: 10 sec between samples for testing
#endif

// Comment out to build the lean receiver without the 24h charts + .bin store
// (equivalent to the old bresser.tc). Defined = the full chart dashboard.
#define WITH_CHARTS 1

#define GDO0_PIN   7

#ifdef WITH_CHARTS
// Ring buffers for the 24h charts — ONE BYTE per sample, not one float.
// Six float[288] cost 6912 bytes of VM heap, claimed as one contiguous block
// when the program loads; as byte[] they cost 1728. On a C3 that block is what
// decides whether the slot starts at all next to WiFi, TLS and the CC1101.
// Each series carries its own scale, because their ranges have nothing in
// common — and the rain step of 0.1 mm is exactly the sensor's own resolution
// (br_rain is a raw count x 0.1), so that series loses nothing at all.
//
// Charted variables: temp, hum, windavg, rain, brightness, uvi
// Stored in a dedicated binary file (CHARTFILE) via save_chart()/load_chart(),
// NOT the .pvs persist store: they are too big to round-trip through
// saveVars() every sample. The file owns header[3] + the 6 arrays.
//
// ⚠️ TEMPERATURE ONLY: raw 0 means "no sample", not 0 °C. The charts are drawn
// over the FULL ring (count = NPTS), so before 24 h of data the slots ahead of
// h_pos are still empty — and with an offset encoding they would decode to
// -30 °C and drag the axis down. Raw 0 is therefore reserved and nulled out in
// the browser, which also fixes the old behaviour where an empty slot drew a
// flat 0 °C line that looked like a real reading. The other five series keep
// 0 = zero: no rain, no wind, dark, no UV are all real values.
#define H_T_SCALE   0.6    // temperature: 0.6 K per step ...
#define H_T_OFFS  -30.0    // ... raw 1 = -29.4 C, raw 255 = +123.0 C
#define H_T_NONE  -30.0    // what raw 0 decodes to = the gap sentinel
#define H_H_SCALE   0.4    // humidity      0..102 %      in 0.4 %
#define H_W_SCALE   0.2    // wind average  0..51 m/s     in 0.2 m/s
#define H_R_SCALE   0.1    // rain/interval 0..25.5 mm    in 0.1 mm (= sensor resolution)
#define H_L_SCALE   0.5    // brightness    0..127.5 kLux in 0.5 kLux
#define H_U_SCALE   0.05   // UV index      0..12.75      in 0.05

byte h_temp[NPTS];
byte h_hum[NPTS];
byte h_wavg[NPTS];
byte h_rain[NPTS];
byte h_lux[NPTS];
byte h_uvi[NPTS];
int h_pos = 0;       // ring buffer write position
int h_count = 0;     // total samples stored (saturates at NPTS)
int sec_cnt = 0;     // seconds counter for store interval
int chdr[3];         // file header: [magic, h_pos, h_count]
#endif

// Raw packet buffer and SPI buffer

// --- CC1101 Low-Level SPI ---

void cc_write_reg(int addr, int val) {
    spi[0] = addr;
    spi[1] = val;
    spiTransfer(1, spi, 2, 1);
}

int cc_read_reg(int addr) {
    spi[0] = addr | 0x80;
    spi[1] = 0;
    spiTransfer(1, spi, 2, 1);
    return spi[1];
}

int cc_read_status(int addr) {
    spi[0] = addr | 0xC0;
    spi[1] = 0;
    spiTransfer(1, spi, 2, 1);
    return spi[1];
}

void cc_strobe(int cmd) {
    spi[0] = cmd;
    spiTransfer(1, spi, 1, 1);
}

void cc_read_burst(int addr, int len) {
    int i;
    spi[0] = addr | 0xC0;
    for (i = 1; i <= len; i++) {
        spi[i] = 0;
    }
    spiTransfer(1, spi, len + 1, 1);
}

void cc_reset() {
    digitalWrite(CS_PIN, 1);
    delayMicroseconds(5);
    digitalWrite(CS_PIN, 0);
    delayMicroseconds(10);
    digitalWrite(CS_PIN, 1);
    delayMicroseconds(41);
    cc_strobe(CMD_SRES);
    delay(2);
}

// --- RSSI Calculation ---

float cc_get_rssi() {
    int raw = cc_read_status(REG_RSSI);
    int rssi;
    if (raw >= 128) {
        rssi = (raw - 256) / 2 - 74;
    } else {
        rssi = raw / 2 - 74;
    }
    return (float)rssi;
}

// --- Bresser 5-in-1 Decoder ---

int decode_5in1() {
    int i;
    for (i = 0; i < 13; i++) {
        if (((msg[i] ^ msg[i + 13]) & 0xFF) != 0xFF) {
            return 0;
        }
    }
    int bits_set = 0;
    int expected = msg[13] & 0xFF;
    for (i = 14; i < DATA_SIZE; i++) {
        int b = msg[i] & 0xFF;
        while (b) {
            bits_set = bits_set + (b & 1);
            b = b >> 1;
        }
    }
    if (bits_set != expected) return 0;

    br_id = msg[14] & 0xFF;
    if (msg[25] & 0x80) { br_batt = 0; } else { br_batt = 1; }

    int temp_raw = (msg[20] & 0x0F) + ((msg[20] >> 4) & 0x0F) * 10 + (msg[21] & 0x0F) * 100;
    if (msg[25] & 0x0F) { temp_raw = 0 - temp_raw; }
    br_temp = (float)temp_raw * 0.1;
    br_hum = (msg[22] & 0x0F) + ((msg[22] >> 4) & 0x0F) * 10;

    int wdir_raw = ((msg[17] >> 4) & 0x0F) * 225;
    int gust_raw = ((msg[17] & 0x0F) << 8) + (msg[16] & 0xFF);
    int wind_raw = (msg[18] & 0x0F) + ((msg[18] >> 4) & 0x0F) * 10 + (msg[19] & 0x0F) * 100;
    br_wind_dir = (float)wdir_raw * 0.1;
    br_wind_gust = (float)gust_raw * 0.1;
    br_wind_avg = (float)wind_raw * 0.1;

    int rain_raw = (msg[23] & 0x0F) + ((msg[23] >> 4) & 0x0F) * 10 + (msg[24] & 0x0F) * 100 + ((msg[24] >> 4) & 0x0F) * 1000;
    br_rain = (float)rain_raw * 0.1;
    return 1;
}

// --- Bresser 6-in-1 Decoder ---

int lfsr_digest16(int start, int bytes, int gen, int key) {
    int sum = 0;
    int k;
    int i;
    for (k = 0; k < bytes; k++) {
        int data = msg[start + k] & 0xFF;
        for (i = 7; i >= 0; i--) {
            if ((data >> i) & 1) { sum = sum ^ key; }
            if (key & 1) { key = (key >> 1) ^ gen; } else { key = key >> 1; }
        }
    }
    return sum;
}

int add_bytes(int start, int num) {
    int result = 0;
    int i;
    for (i = 0; i < num; i++) {
        result = result + (msg[start + i] & 0xFF);
    }
    return result;
}

int decode_6in1() {
    int chkdgst = ((msg[0] & 0xFF) << 8) | (msg[1] & 0xFF);
    int digest = lfsr_digest16(2, 15, 0x8810, 0x5412);
    if (chkdgst != digest) return 0;

    int sum = add_bytes(2, 16);
    if ((sum & 0xFF) != 0xFF) return 0;

    int s_type = (msg[6] >> 4) & 0x0F;
    int id_tmp = ((msg[2] & 0xFF) << 8) | (msg[3] & 0xFF);
    int batt_tmp = (msg[13] >> 1) & 1;
    int flags = msg[16] & 0x0F;

    if (s_type == 4) {
        soil_id = id_tmp;
        soil_batt = batt_tmp;
        soil_rssi = br_rssi;
        int sign = (msg[13] >> 3) & 1;
        int temp_raw = ((msg[12] >> 4) & 0x0F) * 100 + (msg[12] & 0x0F) * 10 + ((msg[13] >> 4) & 0x0F);
        if (sign) { temp_raw = temp_raw - 1000; }
        soil_temp = (float)temp_raw * 0.1;
        int hum_idx = ((msg[14] >> 4) & 0x0F) * 10 + (msg[14] & 0x0F);
        if (hum_idx >= 1 && hum_idx <= 16) {
            if (hum_idx <= 1) { soil_moisture = 0; }
            else if (hum_idx >= 16) { soil_moisture = 99; }
            else { soil_moisture = ((hum_idx - 1) * 100 + 8) / 15; }
        } else { soil_moisture = 0; }
        return 2;
    }

    br_id = id_tmp;
    br_batt = batt_tmp;
    if (flags == 0) {
        int sign = (msg[13] >> 3) & 1;
        int temp_raw = ((msg[12] >> 4) & 0x0F) * 100 + (msg[12] & 0x0F) * 10 + ((msg[13] >> 4) & 0x0F);
        if (sign) { temp_raw = temp_raw - 1000; }
        br_temp = (float)temp_raw * 0.1;
        br_hum = ((msg[14] >> 4) & 0x0F) * 10 + (msg[14] & 0x0F);
    }

    int im7 = (msg[7] ^ 0xFF) & 0xFF;
    int im8 = (msg[8] ^ 0xFF) & 0xFF;
    int im9 = (msg[9] ^ 0xFF) & 0xFF;
    if (im7 <= 0x99 && im8 <= 0x99 && im9 <= 0x99) {
        int gust_raw = ((im7 >> 4) & 0x0F) * 100 + (im7 & 0x0F) * 10 + ((im8 >> 4) & 0x0F);
        int wavg_raw = ((im9 >> 4) & 0x0F) * 100 + (im9 & 0x0F) * 10 + (im8 & 0x0F);
        int wdir_raw = ((msg[10] >> 4) & 0x0F) * 100 + (msg[10] & 0x0F) * 10 + ((msg[11] >> 4) & 0x0F);
        br_wind_gust = (float)gust_raw * 0.1;
        br_wind_avg = (float)wavg_raw * 0.1;
        br_wind_dir = (float)wdir_raw;
    }

    if (flags == 1) {
        int im12 = (msg[12] ^ 0xFF) & 0xFF;
        int im13 = (msg[13] ^ 0xFF) & 0xFF;
        int im14 = (msg[14] ^ 0xFF) & 0xFF;
        int rain_raw = ((im12 >> 4) & 0x0F) * 100000 + (im12 & 0x0F) * 10000 + ((im13 >> 4) & 0x0F) * 1000 + (im13 & 0x0F) * 100 + ((im14 >> 4) & 0x0F) * 10 + (im14 & 0x0F);
        br_rain = (float)rain_raw * 0.1;
    }

    // UV Index (6-in-1): inverted bytes 15-16, BCD, ×0.1
    int im15 = (msg[15] ^ 0xFF) & 0xFF;
    int im16 = (msg[16] ^ 0xFF) & 0xFF;
    if ((msg[16] & 0x0F) == 0 && im15 <= 0x99 && (im16 & 0xF0) <= 0x90) {
        int uv_raw = ((im15 >> 4) & 0x0F) * 100 + (im15 & 0x0F) * 10 + ((im16 >> 4) & 0x0F);
        br_uvi = (float)uv_raw * 0.1;
    }

    return 1;
}

// --- Bresser 7-in-1 Decoder ---

char msgw[DATA_SIZE];

int decode_7in1() {
    int i;
    for (i = 0; i < DATA_SIZE; i++) {
        msgw[i] = msg[i] ^ 0xAA;
    }

    int chkdgst = ((msgw[0] & 0xFF) << 8) | (msgw[1] & 0xFF);
    int digest = 0;
    int key = 0xBA95;
    int gen = 0x8810;
    int k;
    for (k = 0; k < 23; k++) {
        int data = msgw[k + 2] & 0xFF;
        for (i = 7; i >= 0; i--) {
            if ((data >> i) & 1) { digest = digest ^ key; }
            if (key & 1) { key = (key >> 1) ^ gen; } else { key = key >> 1; }
        }
    }
    if ((chkdgst ^ digest) != 0x6DF1) return 0;

    br_id = ((msgw[2] & 0xFF) << 8) | (msgw[3] & 0xFF);
    int s_type = (msg[6] >> 4) & 0x0F;
    int flags = msgw[15] & 0x0F;
    if ((flags & 0x06) == 0x06) { br_batt = 0; } else { br_batt = 1; }

    if (s_type == 1) {
        int wdir = ((msgw[4] >> 4) & 0x0F) * 100 + (msgw[4] & 0x0F) * 10 + ((msgw[5] >> 4) & 0x0F);
        int wgst = ((msgw[7] >> 4) & 0x0F) * 100 + (msgw[7] & 0x0F) * 10 + ((msgw[8] >> 4) & 0x0F);
        int wavg = (msgw[8] & 0x0F) * 100 + ((msgw[9] >> 4) & 0x0F) * 10 + (msgw[9] & 0x0F);
        int rain_raw = ((msgw[10] >> 4) & 0x0F) * 100000 + (msgw[10] & 0x0F) * 10000 + ((msgw[11] >> 4) & 0x0F) * 1000 + (msgw[11] & 0x0F) * 100 + ((msgw[12] >> 4) & 0x0F) * 10 + (msgw[12] & 0x0F);
        int temp_raw = ((msgw[14] >> 4) & 0x0F) * 100 + (msgw[14] & 0x0F) * 10 + ((msgw[15] >> 4) & 0x0F);
        int humidity = ((msgw[16] >> 4) & 0x0F) * 10 + (msgw[16] & 0x0F);
        float temp_c = (float)temp_raw * 0.1;
        if (temp_raw > 600) { temp_c = (float)(temp_raw - 1000) * 0.1; }
        br_temp = temp_c;
        br_hum = humidity;
        br_wind_dir = (float)wdir;
        br_wind_gust = (float)wgst * 0.1;
        br_wind_avg = (float)wavg * 0.1;
        br_rain = (float)rain_raw * 0.1;

        // Light — 6 BCD digits (lux) in de-whitened bytes 17-19, stored as kLux
        int lght_raw = ((msgw[17] >> 4) & 0x0F) * 100000 + (msgw[17] & 0x0F) * 10000 + ((msgw[18] >> 4) & 0x0F) * 1000 + (msgw[18] & 0x0F) * 100 + ((msgw[19] >> 4) & 0x0F) * 10 + (msgw[19] & 0x0F);
        br_lux = (float)lght_raw * 0.001;   // kLux

        // UV Index — 3 BCD digits in de-whitened bytes 20-21, ×0.1
        int uv_raw = ((msgw[20] >> 4) & 0x0F) * 100 + (msgw[20] & 0x0F) * 10 + ((msgw[21] >> 4) & 0x0F);
        br_uvi = (float)uv_raw * 0.1;

        return 1;
    }

    if (s_type == 4) {
        int temp_raw = (msg[20] & 0x0F) + ((msg[20] >> 4) & 0x0F) * 10 + (msg[21] & 0x0F) * 100;
        if (msg[25] & 0x0F) { temp_raw = 0 - temp_raw; }
        soil_temp = (float)temp_raw * 0.1;
        int hum_idx = (msg[22] & 0x0F) + ((msg[22] >> 4) & 0x0F) * 10;
        if (hum_idx >= 1 && hum_idx <= 16) {
            if (hum_idx <= 1) { soil_moisture = 0; }
            else if (hum_idx >= 16) { soil_moisture = 99; }
            else { soil_moisture = ((hum_idx - 1) * 100 + 8) / 15; }
        } else { soil_moisture = 0; }
        soil_id = br_id;
        soil_batt = br_batt;
        soil_rssi = br_rssi;
        return 2;
    }
    return 0;
}

// --- Message Dispatcher ---

int decode_message() {
    int res;
    res = decode_7in1();
    if (res) return res;
    res = decode_6in1();
    if (res) return res;
    res = decode_5in1();
    if (res) return 1;
    return 0;
}

// --- Receive Message ---

int receive_message() {
    int rxBytes = cc_read_status(REG_RXBYTES);
    if (rxBytes >= MSG_BUF_SIZE) {
        cc_read_burst(REG_FIFO, MSG_BUF_SIZE);
        br_rssi = cc_get_rssi();
        cc_strobe(CMD_SIDLE);
        cc_strobe(CMD_SFRX);
        cc_strobe(CMD_SRX);
        delay(1);
        int i;
        for (i = 0; i < DATA_SIZE; i++) {
            msg[i] = spi[i + 2];
        }
        return decode_message();
    }
    return 0;
}

// --- Callbacks ---

#ifndef SIMULATE
void Every50ms() {
    int res = receive_message();
    if (res == 1) {
        br_ok = 1;
        br_cnt = br_cnt + 1;
    }
    if (res == 2) {
        soil_ok = 1;
        soil_cnt = soil_cnt + 1;
    }
}
#endif

#ifdef WITH_CHARTS
// --- Chart persistence: own a dedicated binary file (not .pvs) ---
// Layout: int header[3] = {magic, h_pos, h_count}, then the 6 byte[NPTS]
// arrays back to back. ⚠️ fileReadBin/fileWriteBin move 4-byte int32 elements
// for an int[] but RAW BYTES for a byte[], one file byte per element — which
// is why the header still counts 4 bytes each and the arrays only one.
// Size = 3*4 + 6*NPTS = 1740 bytes (was 6924 with float arrays).
#define CHARTFILE "/bresser_chart.bin"
#define CHART_MAGIC 0x42430103   // bump when stored layout/units change -> old file ignored (0103: byte-packed samples)

void save_chart() {
    int h = fileOpen(CHARTFILE, 1);          // 1 = write (truncate)
    if (h < 0) return;
    chdr[0] = CHART_MAGIC; chdr[1] = h_pos; chdr[2] = h_count;
    fileWriteBin(h, chdr, 3);
    fileWriteBin(h, h_temp, NPTS);
    fileWriteBin(h, h_hum,  NPTS);
    fileWriteBin(h, h_wavg, NPTS);
    fileWriteBin(h, h_rain, NPTS);
    fileWriteBin(h, h_lux,  NPTS);
    fileWriteBin(h, h_uvi,  NPTS);
    fileClose(h);
}

void load_chart() {
    int want = 12 + 6 * NPTS;                // 3 ints + 6 byte arrays
    if (fileSize(CHARTFILE) != want) return; // missing/old-format -> fresh start
    int h = fileOpen(CHARTFILE, 0);          // 0 = read
    if (h < 0) return;
    fileReadBin(h, chdr, 3);
    if (chdr[0] != CHART_MAGIC) { fileClose(h); return; }  // version bump -> ignore old data
    h_pos = chdr[1]; h_count = chdr[2];
    fileReadBin(h, h_temp, NPTS);
    fileReadBin(h, h_hum,  NPTS);
    fileReadBin(h, h_wavg, NPTS);
    fileReadBin(h, h_rain, NPTS);
    fileReadBin(h, h_lux,  NPTS);
    fileReadBin(h, h_uvi,  NPTS);
    fileClose(h);
}

// Encode a value into a raw 0..255 byte. The clamp is not decoration: a byte
// store keeps only the lowest byte, so an out-of-range reading would wrap to a
// plausible-looking wrong value instead of an obvious spike.
int h_enc(float v, float scale, float offs) {
    int raw = (int)((v - offs) / scale + 0.5);
    if (raw < 0)   { raw = 0; }
    if (raw > 255) { raw = 255; }
    return raw;
}

// Temperature: same, but raw 0 is reserved for "no sample" (see the note at
// the array declarations), so a real reading never encodes to 0.
int h_enc_t(float v) {
    int raw = h_enc(v, H_T_SCALE, H_T_OFFS);
    if (raw < 1) { raw = 1; }
    return raw;
}

void store_sample() {
    h_temp[h_pos] = h_enc_t(br_temp);
    h_hum[h_pos] = h_enc((float)br_hum, H_H_SCALE, 0.0);
    h_wavg[h_pos] = h_enc(br_wind_avg, H_W_SCALE, 0.0);
    // rain chart = delta since the last sample (rainfall per interval), not the
    // running total. rain_prev < 0 = first-ever sample -> 0 (no startup spike);
    // br_rain < rain_prev = sensor counter reset -> clamp to 0.
    float rd = 0.0;
    if (rain_prev >= 0.0 && br_rain >= rain_prev) { rd = br_rain - rain_prev; }
    h_rain[h_pos] = h_enc(rd, H_R_SCALE, 0.0);
    rain_prev = br_rain;
    h_lux[h_pos] = h_enc(br_lux, H_L_SCALE, 0.0);
    h_uvi[h_pos] = h_enc(br_uvi, H_U_SCALE, 0.0);
    h_pos++;
    if (h_pos >= NPTS) h_pos = 0;
    if (h_count < NPTS) h_count++;
    // Flush the chart arrays to their own binary file after every sample
    save_chart();
}
#endif

int sim_tick = 0;  // simulation counter

void EverySecond() {
#ifdef SIMULATE
    // Generate fake sensor data with realistic variations
    if (!br_ok) {
        br_ok = 1;
        br_id = 0x1234;
        br_batt = 1;
        soil_ok = 1;
        soil_id = 0x5678;
        soil_batt = 1;
    }
    sim_tick++;
    br_temp = 18.0 + (float)(sim_tick % 60) * 0.1 - 3.0;
    br_hum = 55 + (sim_tick % 30);
    br_wind_avg = 1.0 + (float)(sim_tick % 20) * 0.2;
    br_rain = 0.0 + (float)(sim_tick % 100) * 0.1;
    // kLux, not lux: the real decoder is raw * 0.001 (line ~476) and a bright
    // day is around 100. The old 5000..15000 was a raw-lux figure in a kLux
    // variable — harmless while the chart stored floats, but it now clamps
    // against the byte encoding and hid the real range from the simulation.
    br_lux = 5.0 + (float)(sim_tick % 200) * 0.5;
    br_uvi = (float)(sim_tick % 10) * 0.5;
    br_rssi = -60.0;
    br_cnt = sim_tick;
    soil_temp = 12.0 + (float)(sim_tick % 40) * 0.05;
    soil_moisture = 30 + (sim_tick % 40);
    soil_cnt = sim_tick;
#endif

    if (!br_ok) return;

    // Auto-sent via UDP on assignment (global float)
    rtemp = br_temp;   // current temperature
    train = br_rain;   // total rain (cumulative mm)
    // rain today (0-24h, since local midnight) — only once NTP time is valid
    char rts[32]; timeStamp(rts);
    int rday = timeToSecs(rts) / 86400;
    if (rday > 18000) {
        if (br_rain < rain_base) { rain_base = br_rain; }   // sensor counter reset
        if (rain_day != rday) { rain_base = br_rain; rain_day = rday; }
        rain_today = br_rain - rain_base;
    }
#ifdef WITH_CHARTS
    // average temperature over the day 0-24h = mean of stored chart samples
    // ⚠️ h_temp holds RAW BYTES now — decode before averaging, and skip the
    // reserved 0 (no sample) so a half-filled ring does not drag the mean down.
    float st = 0.0; int jj; int nt = 0;
    for (jj = 0; jj < h_count; jj++) {
        if (h_temp[jj] > 0) { st = st + ((float)h_temp[jj] * H_T_SCALE + H_T_OFFS); nt = nt + 1; }
    }
    if (nt > 0) { avgt = st / (float)nt; }

    // Store first sample immediately so charts appear on first page load
    if (h_count == 0) {
        store_sample();
        sec_cnt = 0;
        return;
    }

    sec_cnt++;
    if (sec_cnt >= STORE_INTERVAL) {
        sec_cnt = 0;
        store_sample();
    }
#endif
}

// WebChart(type, title, unit, color, pos, count, array, decimals, interval, ymin, ymax)
// type: 0=line, 1=column
// Empty title "" = add series to previous chart
// decimals = number of decimal places to display
// interval = minutes between samples (for X-axis time)
// ymin, ymax = Y-axis range (if ymin >= ymax → auto-scale)
void WebPage() {
    if (h_count < 1) {
        webSend("<p>Collecting data... Charts appear after first sample.</p>");
        return;
    }
    // Alle Diagramme in einem Fieldset ("Karte"). Breite 0 = volle
    // Containerbreite, damit sie mit jedem anderen Skript auf der Seite
    // fluchten — eine feste px-Breite wirkt ausserdem auf die NACHFOLGENDEN
    // Skripte weiter (mi-hol, #111). Deshalb auch kein margin-left:-20px
    // mehr: der Ausgleich war nur fuer die feste Breite noetig.
    WebChartSize(0, 280);
    webSend("<fieldset style='border:1px solid rgba(150,150,150,.4);border-radius:8px;margin:8px 2px;padding:4px 9px 8px'><legend style='font-weight:bold;padding:0 6px'>Wetterverlauf 24h</legend><div>");
    // Every series carries its own WebChartQ — it applies to the NEXT WebChart
    // only and is cleared by it, so each one has to be repeated. The scale is
    // what turns the stored byte back into a physical value.
    WebChartQ(H_T_SCALE, H_T_OFFS);
    WebChart(0, "Temperature (24h)", "°C", 0xe74c3c, h_pos, NPTS, h_temp, 1, 5, -20, 50);
    WebChartJS("for(var r=0;r<dt.getNumberOfRows();r++)if(dt.getValue(r,1)==-30)dt.setValue(r,1,null);");   // -30 = H_T_NONE: slot not filled yet -> gap, not a flat 0 C line

    WebChartQ(H_H_SCALE, 0.0);
    WebChart(0, "Humidity (24h)",    "%",        0x3498db, h_pos, NPTS, h_hum,  1, 5, 0, 100);

    WebChartQ(H_W_SCALE, 0.0);
    WebChart(0, "Wind Average (24h)","m/s",      0x27ae60, h_pos, NPTS, h_wavg, 1, 5, 0, 0);
    WebChartJS("o.vAxis=o.vAxis||{};var mx=0;for(var r=0;r<dt.getNumberOfRows();r++){var v=dt.getValue(r,1);if(v>mx)mx=v;}o.vAxis.viewWindow={min:0,max:(mx<2?2:mx*1.15)};");   // wind>=0: floor 0, top=1.15x peak (>=2) so the axis never collapses to a negative default

    WebChartQ(H_R_SCALE, 0.0);
    WebChart(1, "Rain (24h)",        "mm",       0x2980b9, h_pos, NPTS, h_rain, 1, 5, 0, 0);   // type 1 = column (per-interval rainfall)

    WebChartQ(H_L_SCALE, 0.0);
    WebChart(0, "Brightness (24h)",  "kLux",     0xf39c12, h_pos, NPTS, h_lux,  1, 5, 0, 0);

    WebChartQ(H_U_SCALE, 0.0);
    WebChart(0, "UV Index (24h)",    "UVI",      0x9b59b6, h_pos, NPTS, h_uvi,  1, 5, 0, 12);
    webSend("</div></fieldset>");
}

void WebCall() {
    byte out[80];        // Textpuffer: byte[] ist ein vollwertiger Ersatz
    if (br_ok) {
        webSend("{s}{m}{e}<tr><td colspan=2 style='background:#2980b9;color:#fff;padding:6px 10px;font-weight:bold;text-align:center;border-radius:4px'>Wetterstation</td></tr>");
        sprintf(out, "{s}Bresser ID{m}0x%04X{e}", br_id);
        webSend(out);
        LGetString(0, br_lbl);
        sprintf(out, "{s}%s{m}", br_lbl);
        webSend(out);
        sprintf(out, "%.1f °C{e}", br_temp);
        webSend(out);
        LGetString(1, br_lbl);
        sprintf(out, "{s}%s{m}", br_lbl);
        webSend(out);
        sprintf(out, "%d %%{e}", br_hum);
        webSend(out);
        sprintf(out, "{s}Wind Avg{m}%.1f m/s{e}", br_wind_avg);
        webSend(out);
        sprintf(out, "{s}Wind Max{m}%.1f m/s{e}", br_wind_gust);
        webSend(out);
        sprintf(out, "{s}Wind Dir{m}%.0f°{e}", br_wind_dir);
        webSend(out);
        sprintf(out, "{s}Rain{m}%.1f mm{e}", br_rain);
        webSend(out);
        sprintf(out, "{s}Rain today{m}%.1f mm{e}", rain_today);
        webSend(out);
        LGetString(15, br_lbl);
        sprintf(out, "{s}%s{m}", br_lbl);
        webSend(out);
        sprintf(out, "%.1f kLux{e}", br_lux);
        webSend(out);
        sprintf(out, "{s}UV Index{m}%.1f{e}", br_uvi);
        webSend(out);
        sprintf(out, "{s}RSSI{m}%.0f dBm{e}", br_rssi);
        webSend(out);
        if (br_batt) { webSend("{s}Battery{m}<span style='color:#27ae60;font-weight:bold'>OK</span>{e}"); }
        else { webSend("{s}Battery{m}<span style='color:#e74c3c;font-weight:bold'>Low</span>{e}"); }
        sprintf(out, "{s}Packets{m}%d{e}", br_cnt);
        webSend(out);
#ifdef WITH_CHARTS
        sprintf(out, "{s}Avg Temp 0-24h{m}%.1f °C{e}", avgt);
        webSend(out);
#endif
    } else {
        webSend("{s}Bresser{m}waiting for data{e}");
    }
    if (soil_ok) {
        webSend("{s}{m}{e}<tr><td colspan=2 style='background:#8e6b3f;color:#fff;padding:6px 10px;font-weight:bold;text-align:center;border-radius:4px'>Bodensensor</td></tr>");
        sprintf(out, "{s}Soil ID{m}0x%04X{e}", soil_id);
        webSend(out);
        LGetString(0, br_lbl);
        sprintf(out, "{s}Soil %s{m}", br_lbl);
        webSend(out);
        sprintf(out, "%.1f °C{e}", soil_temp);
        webSend(out);
        LGetString(17, br_lbl);
        sprintf(out, "{s}Soil %s{m}", br_lbl);
        webSend(out);
        sprintf(out, "%d %%{e}", soil_moisture);
        webSend(out);
        if (soil_batt) { webSend("{s}Soil Battery{m}<span style='color:#27ae60;font-weight:bold'>OK</span>{e}"); }
        else { webSend("{s}Soil Battery{m}<span style='color:#e74c3c;font-weight:bold'>Low</span>{e}"); }
        sprintf(out, "{s}Soil RSSI{m}%.0f dBm{e}", soil_rssi);
        webSend(out);
        sprintf(out, "{s}Soil Packets{m}%d{e}", soil_cnt);
        webSend(out);
    }
}

int main() {
#ifndef SIMULATE
    spiInit(-1, -1, -1, SPI_MHZ);
    gpioInit(GDO0_PIN, 0);
    gpioInit(CS_PIN, 1);
    cc_reset();
    spiSetCS(1, CS_PIN);

    int ver = cc_read_status(0x31);
    byte buf[64];        // und kostet ein Viertel der Slots
    sprintf(buf, "CC1101 version: 0x%02X\n", ver);
    printString(buf);

    if (ver == 0 || ver == 0xFF) {
        printStr("ERROR: CC1101 not found\n");
        return 1;
    }

    if (!cc_init_bresser()) {
        printStr("ERROR: CC1101 init failed\n");
        return 1;
    }
#else
    printStr("SIMULATE mode — no CC1101 hardware\n");
#endif

    // Init runtime state (not persisted)
    br_ok = 0;
    br_cnt = 0;
    soil_ok = 0;
    soil_cnt = 0;

#ifdef WITH_CHARTS
    // Chart data + counters live in a dedicated binary file (not .pvs)
    load_chart();
    if (h_count > 0) {
        byte ibuf[64];
        sprintf(ibuf, "Restored %d chart points from file\n", h_count);
        printString(ibuf);
    }
#endif

    printStr("Bresser CC1101 ready\n");
#ifdef WITH_CHARTS
    printStr("Charts: 6 variables, 288 pts each (5-min interval)\n");
#endif
    return 0;
}
